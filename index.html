from datetime import datetime import time import requests from flask import Flask, jsonify, request from flask_cors import CORS app = Flask(__name__) CORS(app) # Erlaubt Cross-Origin-Requests für dein Frontend # In-Memory Cache im VPS-Arbeitsspeicher cache = { "timestamp": 0, "data": None, "lat_rounded": None, "lon_rounded": None, } @app.route("/api/regen", methods=["GET"]) def api_regen(): global cache # Koordinaten aus der Frontend-Anfrage holen try: lat = float(request.args.get("lat", 52.3745)) lon = float(request.args.get("lon", 9.7386)) except ValueError: return jsonify({"error": "Invalid coordinates"}), 400 # Auf 2 Nachkommastellen runden (~100m Raster für den Stadtteil) lat_r = round(lat, 2) lon_r = round(lon, 2) current_time = time.time() cache_duration = 5 * 60 # 5 Minuten Cache # 1. Prüfen, ob der Cache noch gültig ist und derselbe Standort abgefragt wird if ( cache["data"] is not None and (current_time - cache["timestamp"] < cache_duration) and cache["lat_rounded"] == lat_r and cache["lon_rounded"] == lon_r ): print("Liefere Daten aus dem Cache...") return jsonify(cache["data"]) # 2. Frische Daten von Open-Meteo abrufen (Kein API-Key nötig!) print("Frische Daten von Open-Meteo werden abgerufen...") url = "https://api.open-meteo.com/v1/forecast" params = { "latitude": lat, "longitude": lon, "minutely_15": "precipitation", "timezone": "Europe/Berlin", } response = requests.get(url, params=params) if response.status_code == 200: raw_data = response.json() try: minutely_data = raw_data.get("minutely_15", {}) times = minutely_data.get("time", []) precips = minutely_data.get("precipitation", []) forecast_minutely = [] is_raining_now = False for i in range(len(times)): t_str = times[i] val = precips[i] if i < len(precips) else 0.0 if val is None: val = 0.0 # ISO Zeit in Unix Timestamp umwandeln dt_obj = datetime.fromisoformat(t_str) ts = int(dt_obj.timestamp()) forecast_minutely.append({"timestamp": ts, "precipitation": val}) # Prüfen ob es im ersten Moment regnet if i == 0 and val >= 0.1: is_raining_now = True formatted_data = { "high_precision": { "rain_now": is_raining_now, "niederschlags_art": "Regen", "forecast_minutely": forecast_minutely, } } # In den Cache schreiben cache["timestamp"] = current_time cache["data"] = formatted_data cache["lat_rounded"] = lat_r cache["lon_rounded"] = lon_r return jsonify(formatted_data) except Exception as e: print(f"Fehler beim Parsen der Open-Meteo Daten: {e}") if cache["data"] is not None: return jsonify(cache["data"]) return jsonify({"error": "Parsing failed", "details": str(e)}), 500 else: print( f"API-Fehler von Open-Meteo: {response.status_code} - {response.text}" ) if cache["data"] is not None: return jsonify(cache["data"]) return jsonify({"error": "API request failed", "details": response.text}), 502 if __name__ == "__main__": app.run(host="0.0.0.0", port=5000)
