from datetime import datetime import time import requests from flask import Flask, jsonify, request from flask_cors import CORS app = Flask(__name__) CORS(app) # Erlaubt Cross-Origin-Requests für dein Frontend # Dein Tomorrow.io API-Key API_KEY = "DEIN_TOMORROW_IO_API_KEY" # In-Memory Cache im VPS-Arbeitsspeicher cache = { "timestamp": 0, "data": None, "lat_rounded": None, "lon_rounded": None, } @app.route("/api/regen", methods=["GET"]) def api_regen(): global cache # Koordinaten aus der Frontend-Anfrage holen try: lat = float(request.args.get("lat", 52.3745)) lon = float(request.args.get("lon", 9.7386)) except ValueError: return jsonify({"error": "Invalid coordinates"}), 400 # Auf 2 Nachkommastellen runden (~100m Raster für den Stadtteil) lat_r = round(lat, 2) lon_r = round(lon, 2) current_time = time.time() cache_duration = 15 * 60 # 15 Minuten Cache # 1. Prüfen, ob der Cache noch gültig ist und derselbe Standort abgefragt wird if ( cache["data"] is not None and (current_time - cache["timestamp"] < cache_duration) and cache["lat_rounded"] == lat_r and cache["lon_rounded"] == lon_r ): print("Liefere Daten aus dem 15-Minuten-Cache...") return jsonify(cache["data"]) # 2. Frische Daten von Tomorrow.io abrufen print("Frische Daten von Tomorrow.io werden abgerufen...") url = "https://api.tomorrow.io/v4/weather/forecast" params = { "location": f"{lat},{lon}", "timesteps": "1m", # Minütliche Intervalle "fields": ["precipitationIntensity", "precipitationType", "temperature"], "units": "metric", "apikey": API_KEY, } response = requests.get(url, params=params) if response.status_code == 200: raw_data = response.json() # Tomorrow.io JSON in das Format mappen, das dein Frontend erwartet try: minutely_intervals = ( raw_data.get("timelines", {}).get("minutely", []) ) forecast_minutely = [] is_raining_now = False max_intensity_now = 0.0 for i, item in enumerate(minutely_intervals[:120]): # 2 Stunden (120 Min) values = item.get("values", {}) precip_intensity = values.get("precipitationIntensity", 0.0) # Convert ISO timestamp to unix timestamp (seconds) dt_obj = datetime.fromisoformat( item.get("time").replace("Z", "+00:00") ) ts = int(dt_obj.timestamp()) forecast_minutely.append( {"timestamp": ts, "precipitation": precip_intensity} ) # Prüfen ob es im ersten Moment (jetzt) regnet if i == 0 and precip_intensity >= 0.1: is_raining_now = True # Niederschlagsart übersetzen (Tomorrow.io gibt Codes/Zahlen oder Strings zurück) # Typischerweise: 0 = Kein, 1 = Regen, 2 = Schnee, 3 = Graupel, 4 = Hagel first_values = ( minutely_intervals[0].get("values", {}) if minutely_intervals else {} ) p_type_code = first_values.get("precipitationType", 0) type_map = { 0: "Kein Niederschlag", 1: "Regen", 2: "Schnee", 3: "Graupel", 4: "Hagel", } niederschlags_art = type_map.get(p_type_code, "Regen") formatted_data = { "high_precision": { "rain_now": is_raining_now, "niederschlags_art": niederschlags_art, "forecast_minutely": forecast_minutely, } } # In den Cache schreiben cache["timestamp"] = current_time cache["data"] = formatted_data cache["lat_rounded"] = lat_r cache["lon_rounded"] = lon_r return jsonify(formatted_data) except Exception as e: print(f"Fehler beim Parsen der Tomorrow.io Daten: {e}") # Fallback auf alten Cache falls vorhanden if cache["data"] is not None: return jsonify(cache["data"]) return jsonify({"error": "Parsing failed", "details": str(e)}), 500 else: # API-Fehler (z.B. Limit erreicht) -> Fallback auf Cache print( f"API-Fehler von Tomorrow.io: {response.status_code} - {response.text}" ) if cache["data"] is not None: print("Liefere alten Cache als Fallback wegen API-Fehler.") return jsonify(cache["data"]) return ( jsonify( {"error": "API request failed", "details": response.text} ), 502, ) if __name__ == "__main__": app.run(host="0.0.0.0", port=5000)
